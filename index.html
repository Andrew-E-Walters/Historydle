// Verifying match updates and styling conditions function submitGuess(guessedFigure) { const row = document.createElement('div'); row.className = 'guess-row'; const matches = { name: guessedFigure.name === secretFigure.name, role: guessedFigure.role === secretFigure.role, domain: guessedFigure.domain === secretFigure.domain, country: guessedFigure.country === secretFigure.country, state: guessedFigure.state === secretFigure.state, gender: guessedFigure.gender === secretFigure.gender, age: guessedFigure.age === secretFigure.age }; // Handle Age arrow tracking directions let ageLabel = guessedFigure.age; let ageClass = "wrong"; if (matches.age) { ageClass = "correct"; } else { ageLabel += guessedFigure.age < secretFigure.age ? " ⬆️" : " ⬇️"; if (Math.abs(guessedFigure.age - secretFigure.age) <= 5) ageClass = "partial"; } // --- FIXED ROW INSERTS HERE --- row.appendChild(createCell(guessedFigure.name, '', matches.name ? 'correct' : 'wrong')); row.appendChild(createCell(guessedFigure.role, '🎭', matches.role ? 'correct' : 'wrong')); row.appendChild(createCell(guessedFigure.domain, domainEmojis[guessedFigure.domain] || '❓', matches.domain ? 'correct' : 'wrong')); row.appendChild(createCell(guessedFigure.country, countryEmojis[guessedFigure.country] || '🏳️', matches.country ? 'correct' : 'wrong')); row.appendChild(createCell(guessedFigure.state, stateEmojis[guessedFigure.state] || '📍', matches.state ? 'correct' : 'wrong')); row.appendChild(createCell(guessedFigure.gender, guessedFigure.gender === "Male" ? "👨" : "👩", matches.gender ? 'correct' : 'wrong')); row.appendChild(createCell(ageLabel, getDecadeEmoji(guessedFigure.age), ageClass)); // Passes variables correctly guessesContainer.insertBefore(row, guessesContainer.firstChild); if (matches.name) { gameOver = true; winMessage.style.display = 'block'; input.disabled = true; } } // --- FIXED CELL BUILDER --- function createCell(text, emoji, matchClass) { const cell = document.createElement('div'); cell.className = `cell ${matchClass}`; if (emoji === '') { cell.className += ' name-cell'; cell.innerText = text; } else { cell.innerHTML = `${emoji}
${text}
`; } return cell; }
